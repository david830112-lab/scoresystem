    // 學生登入處理

    function handleStudentLogin(e) {

    // 渲染教師欄位設定介面

    function renderColumnSettings() {

      const container = document.getElementById('columnSettingsContainer');

      if (gradesData.length === 0) return;


      const keys = Object.keys(gradesData[0]).filter(k => k !== '班級座號');

      let html = '';


      keys.forEach(col => {

        const conf = columnConfigs[col] || { category: '不採計', rawWeight: 10, note: '' };

        html += `

          <div class="grid grid-cols-1 md:grid-cols-4 gap-3 p-3 bg-slate-50 rounded-xl border border-slate-200 items-center">

            <div class="font-semibold text-slate-700 text-sm md:col-span-1">${col}</div>

            <div>

              <select onchange="updateColConfig('${col}', 'category', this.value)" class="w-full text-sm p-2 border rounded-lg">

                <option value="不採計" ${conf.category === '不採計' ? 'selected' : ''}>不採計</option>

                <option value="國語-小考" ${conf.category === '國語-小考' ? 'selected' : ''}>國語 - 小考 (佔20%)</option>

                <option value="國語-作業" ${conf.category === '國語-作業' ? 'selected' : ''}>國語 - 作業 (佔80%)</option>

                <option value="數學-小考" ${conf.category === '數學-小考' ? 'selected' : ''}>數學 - 小考 (佔20%)</option>

                <option value="數學-作業" ${conf.category === '數學-作業' ? 'selected' : ''}>數學 - 作業 (佔80%)</option>

              </select>

            </div>

            <div>

              <input type="number" value="${conf.rawWeight}" min="1" placeholder="配分數字 (如10)" onchange="updateColConfig('${col}', 'rawWeight', parseFloat(this.value)|||0)" class="w-full text-sm p-2 border rounded-lg">

            </div>

            <div>

              <input type="text" value="${conf.note || ''}" placeholder="說明 / 採計備註" onchange="updateColConfig('${col}', 'note', this.value)" class="w-full text-sm p-2 border rounded-lg">

            </div>

          </div>

        `;

      });

      container.innerHTML = html;

    }


    function updateColConfig(col, field, val) {

      if (!columnConfigs[col]) columnConfigs[col] = { category: '不採計', rawWeight: 10, note: '' };

      columnConfigs[col][field] = val;

    }


    // 渲染學生端成績卡片

    function renderStudentDashboard(seatNo) {

      const studentGrades = gradesData.find(g => String(g['班級座號']) === seatNo) || {};

      const container = document.getElementById('subjectCardsContainer');


      const subjects = ['國語', '數學'];

      let html = '';


      subjects.forEach(sub => {

        const quizItems = [];

        const hwItems = [];


        Object.keys(columnConfigs).forEach(col => {

          const cfg = columnConfigs[col];

          if (cfg.category === `${sub}-小考`) quizItems.push({ name: col, ...cfg, score: Number(studentGrades[col]) || 0 });

          if (cfg.category === `${sub}-作業`) hwItems.push({ name: col, ...cfg, score: Number(studentGrades[col]) || 0 });

        });


        // 計算小考權重與加權分數

        const quizTotalRaw = quizItems.reduce((acc, item) => acc + item.rawWeight, 0);

        let quizWeightedScoreSum = 0;

        quizItems.forEach(item => {

          item.pct = quizTotalRaw > 0 ? (item.rawWeight / quizTotalRaw) : 0;

          quizWeightedScoreSum += item.score * item.pct;

        });

        const finalQuizScore = quizWeightedScoreSum * 0.20;


        // 計算作業權重與加權分數

        const hwTotalRaw = hwItems.reduce((acc, item) => acc + item.rawWeight, 0);

        let hwWeightedScoreSum = 0;

        hwItems.forEach(item => {

          item.pct = hwTotalRaw > 0 ? (item.rawWeight / hwTotalRaw) : 0;

          hwWeightedScoreSum += item.score * item.pct;

        });

        const finalHwScore = hwWeightedScoreSum * 0.80;


        const totalScore = Math.min(100, finalQuizScore + finalHwScore);


        html += `

          <div class="bg-white rounded-2xl shadow-sm border border-slate-200 overflow-hidden">

            <div class="bg-indigo-50/50 p-6 border-b border-slate-100 flex justify-between items-center">

              <h3 class="text-xl font-bold text-slate-800">${sub}科</h3>

              <div class="text-right">

                <span class="text-xs text-slate-500 block">科目總分數 (上限 100 分)</span>

                <span class="text-3xl font-extrabold text-indigo-600">${totalScore.toFixed(1)}</span>

                <span class="text-sm font-medium text-slate-600">分</span>

              </div>

            </div>


            <div class="p-6 space-y-6">

              <!-- 小考細項 -->

              <div>

                <h4 class="font-bold text-slate-700 mb-3 flex items-center justify-between">

                  <span>📝 小考項目 (佔 20%)</span>

                  <span class="text-sm text-indigo-600 font-semibold">小考加權分數 (20%): ${finalQuizScore.toFixed(1)} 分</span>

                </h4>

                ${renderTable(quizItems)}

              </div>


              <!-- 作業細項 -->

              <div>

                <h4 class="font-bold text-slate-700 mb-3 flex items-center justify-between">

                  <span>📚 作業項目 (佔 80%)</span>

                  <span class="text-sm text-indigo-600 font-semibold">作業加權分數 (80%): ${finalHwScore.toFixed(1)} 分</span>

                </h4>

                ${renderTable(hwItems)}

              </div>

            </div>

          </div>

        `;

      });


      container.innerHTML = html;

    }


    function renderTable(items) {

      if (items.length === 0) return `<p class="text-sm text-slate-400 italic">尚無相關項目</p>`;

      

      let rows = items.map(item => `

        <tr class="border-b border-slate-100">

          <td class="py-3 px-4">

            <div class="font-medium text-slate-800">${item.name}</div>

            ${item.note ? `<span class="inline-block bg-slate-100 text-slate-600 text-xs px-2 py-0.5 rounded mt-1">${item.note}</span>` : ''}

          </td>

          <td class="py-3 px-4 font-semibold text-slate-700">${item.score} 分</td>

          <td class="py-3 px-4 text-slate-600 font-medium">${(item.pct * 100).toFixed(0)}%</td>

        </tr>

      `).join('');
